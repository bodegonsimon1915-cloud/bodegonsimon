""" Bodegón Simón — Panel Administrativo (escritorio) Ejecutar desde la carpeta del proyecto: python main.py """ import os import sys import traceback import tkinter as tk from tkinter import ttk, messagebox # ── rutas ── BASE = os.path.dirname(os.path.abspath(__file__)) os.chdir(BASE) # ← asegura rutas relativas correctas sys.path.insert(0, BASE) # ══════════ VERIFICACIÓN DE ARCHIVOS ══════════ REQUERIDOS = [ "serviceAccountKey.json", "firebase_client.py", "imagenes.py", "whatsapp.py", os.path.join("subapps", "__init__.py"), os.path.join("subapps", "pedidos_app.py"), os.path.join("subapps", "productos_app.py"), os.path.join("subapps", "pagos_app.py"), os.path.join("subapps", "config_app.py"), ] FALTANTES = [f for f in REQUERIDOS if not os.path.exists(os.path.join(BASE, f))] # ══════════ IMPORTS CON ERROR CLARO ══════════ def _importar_o_abortar(): errores = [] try: import firebase_client except Exception: errores.append("firebase_client.py:\n" + traceback.format_exc()) try: from subapps.pedidos_app import PedidosApp except Exception: PedidosApp = None errores.append("subapps/pedidos_app.py:\n" + traceback.format_exc()) try: from subapps.productos_app import ProductosApp except Exception: ProductosApp = None errores.append("subapps/productos_app.py:\n" + traceback.format_exc()) try: from subapps.pagos_app import PagosApp except Exception: PagosApp = None errores.append("subapps/pagos_app.py:\n" + traceback.format_exc()) try: from subapps.config_app import ConfigApp except Exception: ConfigApp = None errores.append("subapps/config_app.py:\n" + traceback.format_exc()) return firebase_client, PedidosApp, ProductosApp, PagosApp, ConfigApp, errores firebase_client, PedidosApp, ProductosApp, PagosApp, ConfigApp, ERRORES_IMPORT = _importar_o_abortar() class BodegonSimonApp(tk.Tk): def __init__(self): super().__init__() self.title("Bodegón Simón — Panel Administrativo") self.geometry("1180x700") self.minsize(980, 620) # ── avisos de arranque ── problemas = [] if FALTANTES: problemas.append("ARCHIVOS QUE FALTAN:\n• " + "\n• ".join(FALTANTES)) if ERRORES_IMPORT: problemas.append("MÓDULOS CON ERROR:\n\n" + "\n\n".join(ERRORES_IMPORT)) if problemas: messagebox.showwarning( "Bodegón Simón — revisar instalación", "\n\n".join(problemas) + "\n\nLa app abrirá igual; las secciones dañadas no se podrán usar.") # ── conectar Firebase ── if not self._conectar_firebase(): return self._build_ui() self.show(PedidosApp if PedidosApp else ProductosApp) # ─────────── conexión ─────────── def _conectar_firebase(self): try: firebase_client.init() self._project_label = f"🔥 Conectado: {firebase_client.project_id() or '(proyecto)'}" return True except Exception as e: messagebox.showerror("Firebase", f"No se pudo conectar:\n\n{e}") self.destroy() return False # ─────────── interfaz ─────────── def _build_ui(self): self.configure(bg="#f4f6f2") nav = tk.Frame(self, bg="#0b2e1d", width=210) nav.pack(side="left", fill="y") nav.pack_propagate(False) tk.Label(nav, text="🧺 Bodegón Simón", bg="#0b2e1d", fg="white", font=("Segoe UI", 13, "bold")).pack(pady=(18, 2)) tk.Label(nav, text="Panel administrativo", bg="#0b2e1d", fg="#8fbfa3", font=("Segoe UI", 8)).pack(pady=(0, 14)) self.nav_buttons = {} # solo agregamos los botones de módulos que SÍ importaron bien modulos = [("📥 Pedidos", PedidosApp), ("📦 Productos", ProductosApp), ("💳 Pagos", PagosApp), ("⚙️ Configuración", ConfigApp)] for label, cls in modulos: if cls is None: b = tk.Button(nav, text=label + " ⚠️", anchor="w", padx=18, pady=9, bg="#3a1f1f", fg="#ff9c8a", relief="flat", state="disabled", font=("Segoe UI", 10, "bold")) else: b = tk.Button(nav, text=label, anchor="w", padx=18, pady=9, bg="#0b2e1d", fg="#d7f2e2", relief="flat", activebackground="#0a5c2e", activeforeground="white", font=("Segoe UI", 10, "bold"), cursor="hand2", command=lambda c=cls: self.show(c)) b.pack(fill="x", pady=1) self.nav_buttons[cls] = b tk.Label(nav, text=self._project_label, bg="#0b2e1d", fg="#6fae87", font=("Segoe UI", 8), wraplength=180, justify="left").pack(side="bottom", pady=10, padx=8) self.container = tk.Frame(self, bg="#f4f6f2") self.container.pack(side="right", fill="both", expand=True) def show(self, cls): if cls is None: messagebox.showerror("Módulo no disponible", "Este módulo no se pudo cargar.\n" "Revisa el aviso del inicio (archivos faltantes o errores).") return for c, b in self.nav_buttons.items(): if c is not None: b.configure(bg="#0b2e1d", fg="#d7f2e2") self.nav_buttons[cls].configure(bg="#0a5c2e", fg="white") for w in self.container.winfo_children(): w.destroy() try: cls(self.container).pack(fill="both", expand=True) except Exception: # nada falla en silencio: mostramos el error exacto messagebox.showerror("Error al cargar la sección", traceback.format_exc()) if __name__ == "__main__": BodegonSimonApp().mainloop()
